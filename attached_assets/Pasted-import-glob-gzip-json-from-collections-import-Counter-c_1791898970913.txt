import glob, gzip, json
from collections import Counter
clients, targets, protocols, seconds, methods, statuses = [Counter() for _ in range(6)]
for f in sorted(glob.glob("/var/log/caddy/moats-*.log*")):
    print("Reading:", f, flush=True)
    opener = gzip.open if f.endswith(".gz") else open
    with opener(f, "rt", errors="replace") as log:
        for line in log:
            try:
                d = json.loads(line)
            except ValueError:
                continue
            t = float(d.get("ts", 0))
            r = d.get("request") or {}
            if not 1791884400 <= t < 1791884580:
                continue
            if r.get("host", "").split(":")[0] != "pro.moats.app":
                continue
            clients[r.get("client_ip") or r.get("remote_ip") or "unknown"] += 1
            path = r.get("uri", "").split("?")[0]
            target = path if path in ("/", "/index.html", "/robots.txt", "/favicon.ico") else "API" if path.startswith("/api/") else "Assets" if path.startswith("/assets/") else "Other paths"
            targets[target] += 1
            protocols[r.get("proto", "unknown")] += 1
            methods[r.get("method", "unknown")] += 1
            statuses[str(d.get("status", "unknown"))] += 1
            seconds[int(t)] += 1
print("Requests:", sum(clients.values()))
print("Distinct client addresses:", len(clients))
print("Top anonymous client counts:", [n for _, n in clients.most_common(10)])
print("Peak requests logged per second:", max(seconds.values(), default=0))
print("Targets:", dict(targets))
print("Protocols:", dict(protocols))
print("Methods:", dict(methods))
print("Statuses:", dict(statuses))
